/* ---------- RNG déterministe ---------- */
function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
function hash2(x, y, seed) {
  let h = seed ^ Math.imul(x, 374761393) ^ Math.imul(y, 668265263);
  h = Math.imul(h ^ (h >>> 13), 1274126177);
  return ((h ^ (h >>> 16)) >>> 0) / 4294967296;
}

const T = 28;
const SPR_W = 2 * T;
const SPR_H = Math.ceil(Math.sqrt(3) * T);
const CX = SPR_W / 2, CY = SPR_H / 2;

function insideHex(pxl, pyl, margin) {
  const x = Math.abs(pxl - CX + 0.5);
  const y = Math.abs(pyl - CY + 0.5);
  const m = margin || 0;
  if (y > (SQ3 / 2) * (T + m)) return false;
  return SQ3 * x + y <= SQ3 * (T + m);
}

/* Base HD : dégradé vertical doux + éclairage haut-gauche + bruit */
function baseHD(ctx, shades, rng, noiseAmt, grad) {
  const n = shades.length;
  const g = grad === undefined ? 0.6 : grad;
  for (let y = 0; y < SPR_H; y++) {
    for (let x = 0; x < SPR_W; x++) {
      if (!insideHex(x, y, 0.45)) continue;
      let t = 0.55 + (0.5 - y / SPR_H) * g + (1 - x / SPR_W) * 0.08 + (rng() - 0.5) * noiseAmt;
      const idx = Math.max(0, Math.min(n - 1, Math.floor(t * n)));
      px(ctx, x, y, shades[idx]);
    }
  }
}

/* Placement aléatoire garanti dans l'hexagone */
function spotHD(rng, fn, n, pad) {
  let placed = 0, tries = 0;
  while (placed < n && tries < n * 8) {
    tries++;
    const x = Math.floor(3 + rng() * (SPR_W - 6));
    const y = Math.floor(4 + rng() * (SPR_H - 8));
    if (insideHex(x, y, -(pad || 2))) { fn(x, y); placed++; }
  }
}

/* ---- sapin détaillé ---- */
function pineHD(ctx, x, y, h, P, rng) {
  const c = P.canopy, tr = P.trunk;
  const trunkH = Math.max(2, Math.floor(h * 0.18));
  const shW = Math.round(h * 0.16);
  for (let dx = -shW; dx <= shW + 1; dx++)
    if (dx === 0 || dx === 1 || rng() < 0.6) px(ctx, x + dx, y + 1, P.base[1]);
  for (let dy = 0; dy < trunkH; dy++) {
    px(ctx, x, y - dy, tr[1]);
    px(ctx, x + 1, y - dy, tr[0]);
  }
  const top = y - h;
  const cone = h - trunkH;
  for (let row = 0; row < cone; row++) {
    const yy = y - trunkH - row;
    const prog = row / cone;
    let half = Math.max(0, Math.round((1 - prog) * h * 0.28));
    if (row % 4 === 0 && half > 0) half += 1;
    for (let dx = -half; dx <= half; dx++) {
      if (Math.abs(dx) === half && rng() < 0.30) continue;
      let ci;
      if (dx < -half * 0.33) ci = 5 + Math.floor(rng() * 2);
      else if (dx > half * 0.33) ci = 1 + Math.floor(rng() * 2);
      else ci = 3 + Math.floor(rng() * 2);
      if (row % 4 === 3 && rng() < 0.5) ci = Math.max(0, ci - 2);
      px(ctx, x + dx, yy, c[ci]);
    }
  }
  px(ctx, x, top, c[8]); px(ctx, x, top + 1, c[6]);
  px(ctx, x - 1, top + 2, c[7]);
}

/* ---- pic montagneux ---- */
function peakHD(ctx, x, y, w, h, P, rng, snowRatio) {
  const R = P.rock;
  for (let row = 0; row <= h; row++) {
    const prog = row / h;
    const half = Math.round(w * (1 - prog));
    const yy = y - row;
    for (let dx = -half; dx <= half; dx++) {
      const snowLine = 1 - snowRatio;
      const isSnow = snowRatio > 0 &&
        (prog > snowLine + snowRatio * 0.25 || (prog > snowLine && rng() < 0.5));
      if (isSnow) {
        px(ctx, x + dx, yy, dx <= 0 ? P.snow[2] : P.snow[1]);
        continue;
      }
      let ci;
      if (dx < -half * 0.12) ci = 5 + Math.floor(rng() * 2);
      else if (dx > half * 0.12) ci = 1 + Math.floor(rng() * 2);
      else ci = 7;
      px(ctx, x + dx, yy, R[ci]);
    }
  }
  let cx2 = x + Math.floor(w * 0.35);
  for (let row = Math.floor(h * 0.1); row < h * 0.6; row++) {
    const yy = y - row;
    if (rng() < 0.5) cx2 += rng() < 0.5 ? -1 : 0;
    const half = Math.round(w * (1 - row / h));
    if (cx2 > x && cx2 - x < half) px(ctx, cx2, yy, R[0]);
  }
  for (let i = 0; i < w * 1.2; i++) {
    const dx = Math.floor((rng() - 0.5) * w * 2.0);
    px(ctx, x + dx, y + 1 + Math.floor(rng() * 2), rng() < 0.5 ? P.scree : R[2]);
  }
}

/* ---- dune ---- */
function duneHD(ctx, yBase, amp, phase, P, rng) {
  const S = P.base;
  for (let x = 2; x < SPR_W - 2; x++) {
    const yC = Math.floor(yBase + Math.sin((x + phase) * 0.28) * amp);
    if (!insideHex(x, yC, -1.5)) continue;
    px(ctx, x, yC, S[7]);
    const grads = [S[2], S[3], S[4]];
    for (let d = 1; d <= 3; d++) {
      if (!insideHex(x, yC + d, -1.5)) break;
      if (d === 1 || rng() < 0.75 - d * 0.15) px(ctx, x, yC + d, grads[d - 1]);
    }
  }
}
function cactusHD(ctx, x, y, P, rng) {
  const C = P.cactus;
  const h = 8 + Math.floor(rng() * 3);
  for (let dy = 0; dy < h; dy++) {
    px(ctx, x, y - dy, C[2]);
    px(ctx, x - 1, y - dy, C[3]);
    px(ctx, x + 1, y - dy, C[1]);
  }
  const ay = y - Math.floor(h * 0.55);
  px(ctx, x - 2, ay, C[2]); px(ctx, x - 3, ay, C[3]);
  px(ctx, x - 3, ay - 1, C[2]); px(ctx, x - 3, ay - 2, C[3]);
  const by = y - Math.floor(h * 0.4);
  px(ctx, x + 2, by, C[1]); px(ctx, x + 3, by, C[2]);
  px(ctx, x + 3, by - 1, C[1]);
  if (rng() < 0.6) px(ctx, x, y - h, P.flower);
  for (let dx = 0; dx <= 4; dx++) px(ctx, x + 1 + dx, y + 1, P.shade);
}

/* ---- vagues avec écume ---- */
function wavesHD(ctx, P, rng, frame, count, foamChance) {
  const D = P.base;
  for (let i = 0; i < count; i++) {
    const y = Math.floor(5 + rng() * (SPR_H - 10));
    const x0 = Math.floor(3 + rng() * (SPR_W - 16));
    const len = 6 + Math.floor(rng() * 8);
    const off = frame === 1 ? 2 : 0;
    for (let dx = 0; dx < len; dx++) {
      const x = x0 + dx + off;
      const yy = y + Math.floor(Math.sin((dx + i) * 0.8) * 1.2);
      if (!insideHex(x, yy, -2)) continue;
      const edge = dx === 0 || dx === len - 1;
      px(ctx, x, yy, edge ? D[D.length - 3] : D[D.length - 2]);
      if (!edge && rng() < foamChance) px(ctx, x, yy - 1, P.foam[rng() < 0.4 ? 1 : 0]);
    }
  }
  for (let i = 0; i < count * 2; i++) {
    const x = Math.floor(4 + rng() * (SPR_W - 8));
    const y = Math.floor(4 + rng() * (SPR_H - 8));
    if (insideHex(x, y, -2) && rng() < 0.5) px(ctx, x, y, P.sparkle);
  }
}

/* ---- monticule arrondi (collines) ---- */
function moundHD(ctx, x, y, w, h, P, rng) {
  for (let row = 0; row <= h; row++) {
    const prog = row / h;
    const half = Math.round(w * Math.sqrt(Math.max(0, 1 - prog * prog)));
    const yy = y - row;
    for (let dx = -half; dx <= half; dx++) {
      let c;
      if (row === h || (row === h - 1 && Math.abs(dx) < half * 0.5)) c = P.top;
      else if (dx < -half * 0.25) c = P.base[5 + (rng() < 0.4 ? 1 : 0)];
      else if (dx > half * 0.3) c = P.base[1 + (rng() < 0.4 ? 1 : 0)];
      else c = P.base[3 + Math.floor(rng() * 2)];
      px(ctx, x + dx, yy, c);
    }
  }
  for (let dx = -w; dx <= w; dx++)
    if (rng() < 0.7) px(ctx, x + dx, y + 1, P.shadow);
}

/* ---- fissure errante (terres désolées) ---- */
function crackHD(ctx, x0, y0, len, P, rng) {
  let x = x0, y = y0;
  const dirX = rng() < 0.5 ? 1 : -1;
  for (let i = 0; i < len; i++) {
    if (insideHex(x, y, -2)) {
      px(ctx, x, y, P.crack[0]);
      if (rng() < 0.3) px(ctx, x, y + 1, P.crack[1]);
    }
    x += rng() < 0.7 ? dirX : 0;
    y += rng() < 0.55 ? 1 : 0;
    // embranchement
    if (i === Math.floor(len / 2) && rng() < 0.6) {
      let bx = x, by = y;
      for (let j = 0; j < len / 2; j++) {
        bx -= dirX; by += rng() < 0.5 ? 1 : 0;
        if (insideHex(bx, by, -2)) px(ctx, bx, by, P.crack[1]);
      }
    }
  }
}

/* ---- colonne de ruine ---- */
function columnHD(ctx, x, y, h, P, rng) {
  const S = P.stone;
  for (let dy = 0; dy < h; dy++) {
    const broken = dy === h - 1 && rng() < 0.8;
    px(ctx, x, y - dy, S[2]);
    px(ctx, x + 1, y - dy, S[1]);
    if (broken && rng() < 0.5) px(ctx, x + (rng() < 0.5 ? 0 : 1), y - dy, S[3]);
    if (rng() < 0.12) px(ctx, x, y - dy, P.moss);
  }
  // chapiteau (si colonne assez haute)
  if (h > 7) {
    px(ctx, x - 1, y - h + 1, S[3]); px(ctx, x + 2, y - h + 1, S[2]);
  }
  // socle
  px(ctx, x - 1, y, S[1]); px(ctx, x + 2, y, S[0]);
  // ombre portée
  for (let dx = 1; dx <= 3; dx++) px(ctx, x + 2 + dx, y + 1, P.shadow);
}

/* ---- flaque de marécage ---- */
function poolHD(ctx, x, y, w, P, rng) {
  const Q = P.pool;
  for (let dy = -2; dy <= 2; dy++) {
    const half = Math.round(w * Math.sqrt(Math.max(0, 1 - (dy / 2.4) * (dy / 2.4))));
    for (let dx = -half; dx <= half; dx++) {
      if (!insideHex(x + dx, y + dy, -2)) continue;
      const edge = Math.abs(dx) >= half - 1 || Math.abs(dy) === 2;
      px(ctx, x + dx, y + dy, edge ? Q[0] : Q[1]);
    }
  }
  // reflet
  for (let dx = -Math.floor(w * 0.4); dx <= 0; dx++)
    px(ctx, x + dx, y - 1, Q[3]);
  if (rng() < 0.7) px(ctx, x + 1, y, Q[2]);
}

/* ---- roseau à quenouille ---- */
function reedHD(ctx, x, y, P, rng) {
  const h = 5 + Math.floor(rng() * 3);
  for (let dy = 0; dy < h; dy++)
    px(ctx, x, y - dy, dy > h * 0.5 ? P.reed[1] : P.reed[0]);
  px(ctx, x, y - h, P.cattail);
  px(ctx, x, y - h - 1, P.cattail);
  if (rng() < 0.5) px(ctx, x + 1, y - Math.floor(h * 0.6), P.reed[1]);
}

/* ---- coulée de lave ---- */
function lavaFlowHD(ctx, x0, y0, len, P, rng) {
  const L = P.lava;
  let x = x0, y = y0;
  for (let i = 0; i < len; i++) {
    const heat = i / len; // 0 = chaud (haut), 1 = froid (bas)
    const ci = Math.min(L.length - 1, Math.floor(heat * L.length));
    if (insideHex(x, y, -1.5)) {
      px(ctx, x, y, L[ci]);
      if (heat < 0.4 && rng() < 0.5) px(ctx, x + 1, y, L[Math.min(4, ci + 1)]);
    }
    y += 1;
    x += rng() < 0.4 ? (rng() < 0.5 ? 1 : -1) : 0;
  }
}

/* =========================================================
   decorateHD — les 16 terrains
   ========================================================= */
function decorateHD(ctx, terrain, rng, variant, frame) {
  const P = PAL_HD[terrain];
  switch (terrain) {

    case "forest": {
      for (let i = 0; i < 14; i++) {
        const x = Math.floor(rng() * SPR_W), y = Math.floor(rng() * SPR_H);
        if (insideHex(x, y, -2) && rng() < 0.5) px(ctx, x, y, P.light);
      }
      const trees = [];
      let tries = 0;
      while (trees.length < 7 && tries < 80) {
        tries++;
        const x = Math.floor(5 + rng() * (SPR_W - 10));
        const y = Math.floor(10 + rng() * (SPR_H - 14));
        if (!insideHex(x, y, -6)) continue;
        if (trees.some(t => Math.hypot(t.x - x, t.y - y) < 9)) continue;
        trees.push({ x, y, h: 13 + Math.floor(rng() * 7) });
      }
      trees.sort((a, b) => a.y - b.y);
      trees.forEach(t => pineHD(ctx, t.x, t.y, t.h, P, rng));
      for (let i = 0; i < 2; i++) {
        const x = Math.floor(6 + rng() * (SPR_W - 12)), y = Math.floor(SPR_H * 0.6 + rng() * SPR_H * 0.3);
        if (insideHex(x, y, -3)) {
          px(ctx, x, y, P.mushroom); px(ctx, x + 1, y, P.mushroom);
          px(ctx, x, y - 1, P.mushroomDot);
        }
      }
      break;
    }

    case "mountains":
      peakHD(ctx, CX - 11, CY + 13, 14, 24, P, rng, 0.38);
      peakHD(ctx, CX + 13, CY + 15, 11, 19, P, rng, 0.34);
      peakHD(ctx, CX + 1, CY + 19, 10, 13, P, rng, 0.0);
      break;

    case "desert": {
      duneHD(ctx, 13, 2.0, variant * 3, P, rng);
      duneHD(ctx, 26, 2.5, variant * 3 + 9, P, rng);
      duneHD(ctx, 38, 1.8, variant * 3 + 18, P, rng);
      cactusHD(ctx, CX - 8 + Math.floor(rng() * 16), CY + 8 + Math.floor(rng() * 6), P, rng);
      const bx = Math.floor(10 + rng() * (SPR_W - 20)), by = Math.floor(8 + rng() * 8);
      if (insideHex(bx, by, -4)) {
        px(ctx, bx, by, P.bone); px(ctx, bx + 1, by, P.bone); px(ctx, bx + 2, by, P.bone);
        px(ctx, bx, by - 1, P.bone); px(ctx, bx + 3, by - 1, P.bone);
      }
      break;
    }

    case "shallow_water":
      wavesHD(ctx, P, mulberry32(0xEA0 + variant * 733), frame, 7, 0.5);
      break;

    case "deep_water":
      wavesHD(ctx, P, mulberry32(0xEB0 + variant * 733), frame, 4, 0.25);
      break;

    case "fertile_land": {
      // sillons courbés + rangées de pousses
      for (let row = 0; row < 5; row++) {
        const yBase = 7 + row * 8 + (variant % 2) * 2;
        for (let x = 2; x < SPR_W - 2; x++) {
          const y = Math.floor(yBase + Math.sin(x * 0.14 + row) * 1.4);
          if (!insideHex(x, y, -1.5)) continue;
          px(ctx, x, y, P.soil[0]);
          if (rng() < 0.5) px(ctx, x, y + 1, P.soil[1]);
          // pousses sur la rangée intermédiaire
          if (x % 3 === (row % 3) && insideHex(x, y - 3, -1.5)) {
            px(ctx, x, y - 3, P.sprout[0]);
            px(ctx, x, y - 4, P.sprout[1]);
            if (rng() < 0.25) px(ctx, x, y - 5, P.wheatTip);
          }
        }
      }
      break;
    }

    case "plains": {
      // touffes d'herbe hautes /|\
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.tuft[0]); px(ctx, x, y - 1, P.tuft[0]); px(ctx, x, y - 2, P.tuft[1]);
        px(ctx, x - 1, y, P.tuft[0]); px(ctx, x - 1, y - 1, P.tuft[1]);
        px(ctx, x + 1, y, P.tuft[0]); px(ctx, x + 1, y - 1, P.tuft[1]);
      }, 9, 3);
      // fleurs sauvages
      spotHD(rng, (x, y) => {
        const c = rng() < 0.5 ? P.flowerW : P.flowerY;
        px(ctx, x, y, c); px(ctx, x - 1, y, c); px(ctx, x + 1, y, c);
        px(ctx, x, y - 1, c); px(ctx, x, y + 1, c);
        px(ctx, x, y, P.flowerCore);
      }, 4, 3);
      break;
    }

    case "sacred_plains": {
      // menhir doré au centre-lumière
      const mx = CX - 6 + Math.floor(rng() * 12), my = CY + 4 + Math.floor(rng() * 5);
      // halo de lumière au sol
      for (let dy = -4; dy <= 4; dy++) for (let dx = -5; dx <= 5; dx++) {
        if (Math.hypot(dx, dy * 1.3) < 5 && insideHex(mx + dx, my + dy, -2) && rng() < 0.3)
          px(ctx, mx + dx, my + dy, P.glow[0]);
      }
      for (let dy = 0; dy < 9; dy++) {
        const w = dy > 6 ? 0 : 1;
        for (let dx = -w; dx <= w; dx++)
          px(ctx, mx + dx, my - dy, dx < 0 ? P.stone[3] : P.stone[1]);
      }
      px(ctx, mx, my - 4, P.rune); px(ctx, mx, my - 2, P.rune);
      // scintillements en croix
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.glow[1]);
        px(ctx, x - 1, y, P.glow[0]); px(ctx, x + 1, y, P.glow[0]);
        px(ctx, x, y - 1, P.glow[0]); px(ctx, x, y + 1, P.glow[0]);
      }, 3, 3);
      break;
    }

    case "enchanted_meadow": {
      // fleurs en grappes
      spotHD(rng, (x, y) => {
        const c = [P.flowerP, P.flowerB, P.flowerW][Math.floor(rng() * 3)];
        px(ctx, x, y, c); px(ctx, x + 1, y, c); px(ctx, x, y - 1, c);
        px(ctx, x - 1, y + 1, P.base[6]);
      }, 7, 2);
      // lucioles : point brillant + halo
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.fly);
        px(ctx, x - 1, y, P.flyHalo); px(ctx, x + 1, y, P.flyHalo);
        px(ctx, x, y - 1, P.flyHalo); px(ctx, x, y + 1, P.flyHalo);
      }, 4, 3);
      break;
    }

    case "hills": {
      moundHD(ctx, CX - 10, CY + 8, 11, 8, P, rng);
      moundHD(ctx, CX + 11, CY + 12, 9, 7, P, rng);
      moundHD(ctx, CX + 1, CY + 18, 8, 5, P, rng);
      // buissons
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.bush[1]); px(ctx, x + 1, y, P.bush[0]);
        px(ctx, x - 1, y, P.bush[1]); px(ctx, x, y - 1, P.bush[2]);
      }, 3, 4);
      break;
    }

    case "wasteland": {
      crackHD(ctx, Math.floor(8 + rng() * 14), Math.floor(6 + rng() * 8), 14, P, rng);
      crackHD(ctx, Math.floor(SPR_W * 0.55 + rng() * 10), Math.floor(10 + rng() * 10), 12, P, rng);
      // squelette : crâne + côtes
      const sx = Math.floor(12 + rng() * (SPR_W - 24)), sy = Math.floor(SPR_H * 0.55 + rng() * 6);
      if (insideHex(sx, sy, -4)) {
        px(ctx, sx, sy, P.bone); px(ctx, sx + 1, sy, P.bone);
        px(ctx, sx, sy - 1, P.bone); px(ctx, sx + 1, sy - 1, P.boneDim);
        px(ctx, sx, sy + 1, P.boneDim);
        for (let i = 0; i < 4; i++) {
          px(ctx, sx + 3 + i * 2, sy, P.bone);
          px(ctx, sx + 3 + i * 2, sy + 1, P.boneDim);
        }
        for (let i = 0; i < 7; i++) px(ctx, sx + 3 + i, sy - 1, P.boneDim);
      }
      // arbuste mort
      const dx2 = Math.floor(10 + rng() * (SPR_W - 20)), dy2 = Math.floor(8 + rng() * 8);
      if (insideHex(dx2, dy2, -4)) {
        px(ctx, dx2, dy2, P.shrub[0]); px(ctx, dx2, dy2 - 1, P.shrub[0]); px(ctx, dx2, dy2 - 2, P.shrub[1]);
        px(ctx, dx2 - 1, dy2 - 3, P.shrub[1]); px(ctx, dx2 + 1, dy2 - 3, P.shrub[0]);
        px(ctx, dx2 - 2, dy2 - 4, P.shrub[1]);
      }
      break;
    }

    case "ancient_ruins": {
      columnHD(ctx, CX - 10, CY + 4, 9, P, rng);
      columnHD(ctx, CX + 8, CY + 8, 6, P, rng);
      // colonne effondrée : cylindre horizontal segmenté
      const fy = CY + 14;
      for (let dx = 0; dx < 10; dx++) {
        const x = CX - 5 + dx;
        if (!insideHex(x, fy, -3)) continue;
        px(ctx, x, fy - 1, P.stone[3]);
        px(ctx, x, fy, dx % 3 === 0 ? P.stoneDark : P.stone[2]);
        px(ctx, x, fy + 1, P.stone[0]);
      }
      // blocs épars moussus
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.stone[2]); px(ctx, x + 1, y, P.stone[1]);
        px(ctx, x, y - 1, P.stone[3]); px(ctx, x + 1, y - 1, P.stone[2]);
        if (rng() < 0.5) px(ctx, x, y - 1, P.moss);
        px(ctx, x + 2, y + 1, P.shadow);
      }, 3, 3);
      break;
    }

    case "caves": {
      // bouche de caverne en arche
      const mw = 9, mh = 8;
      for (let dy = 0; dy <= mh; dy++) {
        const prog = dy / mh;
        const half = Math.round(mw * Math.sqrt(Math.max(0, 1 - (1 - prog) * (1 - prog))));
        for (let dx = -half; dx <= half; dx++)
          px(ctx, CX + dx, CY - mh + 4 + dy, P.mouth);
      }
      // rebord éclairé de l'arche
      for (let a = 0; a <= 20; a++) {
        const ang = Math.PI * (a / 20);
        const rx = Math.round(Math.cos(ang) * (mw + 1));
        const ry = -Math.round(Math.sin(ang) * mh);
        px(ctx, CX + rx, CY + 4 + ry, a < 12 ? P.rim[1] : P.rim[0]);
      }
      // stalactite et stalagmites
      px(ctx, CX - 2, CY - 2, P.base[4]); px(ctx, CX - 2, CY - 1, P.base[3]);
      px(ctx, CX + 3, CY + 2, P.base[4]); px(ctx, CX + 3, CY + 3, P.base[5]); px(ctx, CX + 3, CY + 4, P.base[5]);
      // cristaux lumineux près de l'entrée
      for (const [gx, gy] of [[CX - mw - 3, CY + 5], [CX + mw + 2, CY + 3]]) {
        if (!insideHex(gx, gy, -3)) continue;
        px(ctx, gx, gy, P.crystal[1]); px(ctx, gx, gy - 1, P.crystal[2]);
        px(ctx, gx + 1, gy, P.crystal[0]);
      }
      break;
    }

    case "tundra": {
      // congères : lignes ondulées blanches
      for (let i = 0; i < 3; i++) {
        const yy = 10 + i * 12 + (variant % 2) * 3;
        for (let x = 2; x < SPR_W - 2; x++) {
          const y = Math.floor(yy + Math.sin((x + i * 7) * 0.22) * 1.8);
          if (!insideHex(x, y, -1.5)) continue;
          px(ctx, x, y, P.driftHi);
          if (rng() < 0.6) px(ctx, x, y + 1, P.driftLo);
        }
      }
      // rochers avec chapeau de neige
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.rock[1]); px(ctx, x + 1, y, P.rock[0]);
        px(ctx, x - 1, y, P.rock[2]); px(ctx, x, y + 1, P.rock[0]);
        px(ctx, x, y - 1, P.driftHi); px(ctx, x + 1, y - 1, P.driftHi);
      }, 3, 3);
      // herbes gelées
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.grass); px(ctx, x, y - 1, P.grass);
        px(ctx, x + 1, y, P.grass);
      }, 3, 2);
      break;
    }

    case "swamp": {
      poolHD(ctx, CX - 8, CY + 2, 7, P, rng);
      poolHD(ctx, CX + 10, CY + 10, 5, P, rng);
      // roseaux en bordure des flaques
      reedHD(ctx, CX - 15, CY + 3, P, rng);
      reedHD(ctx, CX - 2, CY + 5, P, rng);
      reedHD(ctx, CX + 5, CY + 12, P, rng);
      reedHD(ctx, CX + 16, CY + 9, P, rng);
      // brume éparse
      spotHD(rng, (x, y) => {
        px(ctx, x, y, P.mist); px(ctx, x + 1, y, P.mist);
        if (rng() < 0.5) px(ctx, x + 2, y, P.mist);
      }, 4, 2);
      break;
    }

    case "volcano": {
      // cône sombre
      const vw = 15, vh = 22, vy = CY + 14;
      for (let row = 0; row <= vh; row++) {
        const prog = row / vh;
        const half = Math.round(vw * (1 - prog));
        const yy = vy - row;
        for (let dx = -half; dx <= half; dx++) {
          let c;
          if (dx < -half * 0.15) c = P.cone[3 + (rng() < 0.4 ? 1 : 0)];
          else if (dx > half * 0.15) c = P.cone[0 + (rng() < 0.4 ? 1 : 0)];
          else c = P.cone[2];
          px(ctx, CX + dx, yy, c);
        }
      }
      // cratère incandescent
      const cyv = vy - vh;
      px(ctx, CX - 1, cyv + 1, P.lava[1]); px(ctx, CX, cyv + 1, P.lava[0]);
      px(ctx, CX + 1, cyv + 1, P.lava[1]); px(ctx, CX, cyv + 2, P.lava[2]);
      px(ctx, CX - 2, cyv + 2, P.lava[2]); px(ctx, CX + 2, cyv + 2, P.lava[3]);
      // coulées
      lavaFlowHD(ctx, CX - 1, cyv + 3, Math.floor(vh * 0.8), P, rng);
      lavaFlowHD(ctx, CX + 2, cyv + 4, Math.floor(vh * 0.55), P, rng);
      // braises autour
      spotHD(rng, (x, y) => px(ctx, x, y, rng() < 0.5 ? P.lava[1] : P.lava[2]), 4, 3);
      // fumée au-dessus du cratère
      for (let i = 0; i < 5; i++) {
        const sx2 = CX + Math.floor((rng() - 0.3) * 6);
        const sy2 = cyv - 1 - i;
        if (insideHex(sx2, sy2, -1))
          px(ctx, sx2, sy2, P.smoke[Math.min(2, Math.floor(i / 2))]);
      }
      break;
    }
  }
}

/* =========================================================
   Cache de sprites
   ========================================================= */
const spriteCache = new Map();
function getSprite(terrain, variant, frame) {
  const key = terrain + "|" + variant + "|" + frame;
  let c = spriteCache.get(key);
  if (c) return c;
  c = document.createElement("canvas");
  c.width = SPR_W; c.height = SPR_H;
  const ctx = c.getContext("2d");
  const cfg = BASECFG[terrain];
  const rng = mulberry32(0x4D5A + variant * 1409 + terrain.charCodeAt(0) * 257 + terrain.length * 61);
  baseHD(ctx, cfg.shades, rng, cfg.noise, cfg.grad);
  decorateHD(ctx, terrain, rng, variant, frame);
  spriteCache.set(key, c);
  return c;
}

function getFogSprite() {
  let c = spriteCache.get("__fog");
  if (c) return c;
  c = document.createElement("canvas");
  c.width = SPR_W; c.height = SPR_H;
  const ctx = c.getContext("2d");
  const rng = mulberry32(0xF06);
  for (let y = 0; y < SPR_H; y++)
    for (let x = 0; x < SPR_W; x++)
      if (insideHex(x, y, 0.45))
        px(ctx, x, y, rng() < 0.12 ? "#1c1a16" : "#100e0b");
  spriteCache.set("__fog", c);
  return c;
}

      P(2,3,"#9a6bd4"); P(3,3,"#b98fee"); P(4,3,"#7a4cb4");
      P(2,4,"#7a4cb4"); P(3,4,"#9a6bd4"); P(4,4,"#5c3690"); break;
    default:
      P(3,3,"#ffffff");
  }
  spriteCache.set(key, c);
  return c;
}

/* =========================================================
   Génération de carte — archipel déterministe
   ========================================================= */
const MAP_W = 52, MAP_H = 40;
let SEED = 1042;
let tiles = [];

function valueNoise(x, y, seed, scale) {
  const gx = x / scale, gy = y / scale;
  const x0 = Math.floor(gx), y0 = Math.floor(gy);
  const fx = gx - x0, fy = gy - y0;
  const sm = t => t * t * (3 - 2 * t);
  const v00 = hash2(x0, y0, seed),     v10 = hash2(x0 + 1, y0, seed);
  const v01 = hash2(x0, y0 + 1, seed), v11 = hash2(x0 + 1, y0 + 1, seed);
  const ix0 = v00 + (v10 - v00) * sm(fx);
  const ix1 = v01 + (v11 - v01) * sm(fx);
  return ix0 + (ix1 - ix0) * sm(fy);
}

function generateMap(seed) {
  tiles = [];
  const centers = [];
  const rng = mulberry32(seed);
  const nIslands = 5 + Math.floor(rng() * 3);
  for (let i = 0; i < nIslands; i++) {
    centers.push({
      x: 7 + rng() * (MAP_W - 14),
      y: 6 + rng() * (MAP_H - 12),
      r: 8 + rng() * 11
    });
  }
  for (let y = 0; y < MAP_H; y++) {
    const row = [];
    for (let x = 0; x < MAP_W; x++) {
      let elev = 0;
      for (const c of centers) {
        const d = Math.hypot(x - c.x, (y - c.y) * 1.15);
        elev = Math.max(elev, Math.max(0, 1 - d / c.r));
      }
      elev = elev * 0.75 + valueNoise(x, y, seed ^ 0x51, 5.5) * 0.35;
      const moist = valueNoise(x, y, seed ^ 0x9E, 7);
      const temp  = valueNoise(x, y, seed ^ 0x33, 11);
      const rare  = hash2(x, y, seed ^ 0x777);

      let terrain;
      if (elev < 0.30) terrain = "deep_water";
      else if (elev < 0.40) terrain = "shallow_water";
      else if (elev > 0.86) terrain = rare < 0.06 ? "volcano" : "mountains";
      else if (elev > 0.74) terrain = temp < 0.3 ? "tundra" : "hills";
      else if (moist > 0.66) terrain = elev < 0.55 ? "swamp" : "forest";
      else if (moist > 0.55) terrain = "forest";
      else if (temp > 0.68 && moist < 0.42) terrain = "desert";
      else if (moist < 0.3) terrain = "wasteland";
      else if (rare < 0.015) terrain = "ancient_ruins";
      else if (rare < 0.03) terrain = "sacred_plains";
      else if (rare < 0.045) terrain = "enchanted_meadow";
      else if (rare < 0.06) terrain = "caves";
      else terrain = moist > 0.45 ? "fertile_land" : "plains";

      let resource = null;
      const rr = hash2(x, y, seed ^ 0xBEEF);
      if (rr < 0.16) {
        const pools = {
          fertile_land: ["wheat", "herbs"], plains: ["wheat", "deer"],
          forest: ["deer", "herbs"], hills: ["stone", "iron"],
          mountains: ["stone", "gold", "iron"], desert: ["gold"],
          shallow_water: ["fish"], deep_water: ["fish"],
          tundra: ["deer"], swamp: ["herbs"], caves: ["crystals", "iron"],
          ancient_ruins: ["crystals"], sacred_plains: ["crystals"]
        };
        const pool = pools[terrain];
        if (pool) resource = pool[Math.floor(hash2(x, y, seed ^ 0xCAFE) * pool.length)];
      }

      const hasRiver = !WATER_TERRAINS.has(terrain) &&
        terrain !== "mountains" && terrain !== "volcano" &&
        hash2(x, y, seed ^ 0x111) < 0.05;

      row.push({ x, y, terrain, resource, hasRiver });
    }
    tiles.push(row);
  }
}

/* =========================================================
   Rendu carte — même géométrie que GameEngine
   ========================================================= */
const canvas = document.getElementById("map");
const ctx = canvas.getContext("2d");
let hexSize = 44;
let camX = 0, camY = 0;
let selected = null, hovered = null;
let waterFrame = 0;
let fogOn = false, gridOn = true, animOn = true;

function resize() {
  const r = canvas.parentElement.getBoundingClientRect();
  canvas.width = r.width * devicePixelRatio;
  canvas.height = r.height * devicePixelRatio;
  ctx.setTransform(devicePixelRatio, 0, 0, devicePixelRatio, 0, 0);
  render();
}
window.addEventListener("resize", resize);

function hexToScreen(x, y) {
  const hexHeight = hexSize * SQ3;
  return {
    sx: x * hexSize * 1.5 - camX,
    sy: y * hexHeight + (x % 2) * (hexHeight / 2) - camY
  };
}
function screenToHex(sx, sy) {
  const hexHeight = hexSize * SQ3;
  const wx = sx + camX;
  const x = Math.round(wx / (hexSize * 1.5));
  for (let dx = -1; dx <= 1; dx++) {
    for (let dy = -1; dy <= 1; dy++) {
      const hx = x + dx;
      if (hx < 0 || hx >= MAP_W) continue;
      const hy = Math.round(((sy + camY) - (hx % 2) * hexHeight / 2) / hexHeight) + dy;
      if (hy < 0 || hy >= MAP_H) continue;
      const { sx: cx2, sy: cy2 } = hexToScreen(hx, hy);
      if (Math.hypot(sx - cx2, sy - cy2) < hexSize * 0.9) return { x: hx, y: hy };
    }
  }
  return null;
}

function hexPath(cx, cy, size) {
  ctx.beginPath();
  for (let i = 0; i < 6; i++) {
    const a = (i * Math.PI) / 3;
    const hx = cx + size * Math.cos(a);
    const hy = cy + size * Math.sin(a);
    i === 0 ? ctx.moveTo(hx, hy) : ctx.lineTo(hx, hy);
  }
  ctx.closePath();
}

function visionState(x, y) {
  if (!fogOn) return 2;
  const c = selected || { x: Math.floor(MAP_W / 2), y: Math.floor(MAP_H / 2) };
  const d = Math.hypot(x - c.x, y - c.y);
  if (d <= 4.2) return 2;
  if (d <= 6.2) return 1;
  return 0;
}

function render() {
  const W = canvas.width / devicePixelRatio;
  const H = canvas.height / devicePixelRatio;
  ctx.imageSmoothingEnabled = false;
  ctx.fillStyle = "#0c0a08";
  ctx.fillRect(0, 0, W, H);

  const hexHeight = hexSize * SQ3;
  const sprW = hexSize * 2, sprH = SPR_H * (hexSize * 2 / SPR_W);

  const x0 = Math.max(0, Math.floor(camX / (hexSize * 1.5)) - 1);
  const x1 = Math.min(MAP_W - 1, Math.ceil((camX + W) / (hexSize * 1.5)) + 1);
  const y0 = Math.max(0, Math.floor(camY / hexHeight) - 1);
  const y1 = Math.min(MAP_H - 1, Math.ceil((camY + H) / hexHeight) + 1);

  for (let x = x0; x <= x1; x++) {
    for (let y = y0; y <= y1; y++) {
      const tile = tiles[y][x];
      const { sx, sy } = hexToScreen(x, y);
      const vis = visionState(x, y);
      const variant = Math.floor(hash2(x, y, 0x5EED) * 4);

      if (vis === 0) {
        ctx.drawImage(getFogSprite(), sx - sprW / 2, sy - sprH / 2, sprW, sprH);
        continue;
      }

      const frame = (animOn && WATER_TERRAINS.has(tile.terrain)) ? waterFrame : 0;
      ctx.drawImage(getSprite(tile.terrain, variant, frame),
        sx - sprW / 2, sy - sprH / 2, sprW, sprH);

      if (tile.hasRiver && vis >= 1) {
        const rw = Math.max(2, hexSize * 0.14);
        ctx.fillStyle = "#3f8ab4";
        ctx.fillRect(sx - hexSize / 2, sy - rw / 2, hexSize, rw);
        ctx.fillStyle = "#7fc4e2";
        ctx.fillRect(sx - hexSize / 2, sy - rw / 6, hexSize, Math.max(1, rw / 3));
      }

      if (tile.resource && vis === 2) {
        const icon = getResIcon(tile.resource);
        const s = hexSize * 0.55;
        ctx.drawImage(icon, sx - s / 2, sy - s / 2, s, s);
      }

      if (vis === 1) {
        hexPath(sx, sy, hexSize + 0.5);
        ctx.fillStyle = "rgba(12, 10, 8, 0.62)";
        ctx.fill();
      }
    }
  }

  for (let x = x0; x <= x1; x++) {
    for (let y = y0; y <= y1; y++) {
      const { sx, sy } = hexToScreen(x, y);
      if (gridOn) {
        hexPath(sx, sy, hexSize);
        ctx.strokeStyle = "rgba(0,0,0,0.28)";
        ctx.lineWidth = 1;
        ctx.stroke();
      }
    }
  }
  if (hovered) {
    const { sx, sy } = hexToScreen(hovered.x, hovered.y);
    hexPath(sx, sy, hexSize - 1);
    ctx.strokeStyle = "rgba(232, 178, 58, 0.55)";
    ctx.lineWidth = 2;
    ctx.stroke();
  }
  if (selected) {
    const { sx, sy } = hexToScreen(selected.x, selected.y);
    hexPath(sx, sy, hexSize - 1);
    ctx.strokeStyle = "#e8b23a";
    ctx.lineWidth = 3;
    ctx.stroke();
    hexPath(sx, sy, hexSize + 2);
    ctx.strokeStyle = "rgba(0,0,0,0.6)";
    ctx.lineWidth = 1;
    ctx.stroke();
  }
}

/* ---------- HUD ---------- */
function updateHUD(tile) {
  const $ = id => document.getElementById(id);
  if (!tile) {
    $("hudTerrain").textContent = "—"; $("hudCoord").textContent = "—";
    $("hudCost").textContent = "—"; $("hudRes").textContent = "—";
    return;
  }
  const cost = TERRAIN_COSTS[tile.terrain];
  $("hudTerrain").textContent = TERRAIN_FR[tile.terrain] || tile.terrain;
  $("hudCoord").textContent = "(" + tile.x + ", " + tile.y + ")";
  $("hudCost").textContent = cost >= IMPASSABLE ? "Infranchissable" : cost + " PA";
  $("hudRes").textContent = tile.resource ? (RES_FR[tile.resource] || tile.resource) : "aucune";
}

/* ---------- Interactions ---------- */
let dragging = false, lastX = 0, lastY = 0, moved = 0;
canvas.addEventListener("mousedown", e => {
  dragging = true; moved = 0;
  lastX = e.clientX; lastY = e.clientY;
  canvas.classList.add("dragging");
});
window.addEventListener("mouseup", () => {
  dragging = false;
  canvas.classList.remove("dragging");
});
window.addEventListener("mousemove", e => {
  if (dragging) {
    camX -= e.clientX - lastX;
    camY -= e.clientY - lastY;
    moved += Math.abs(e.clientX - lastX) + Math.abs(e.clientY - lastY);
    lastX = e.clientX; lastY = e.clientY;
    render();
  } else {
    const r = canvas.getBoundingClientRect();
    const h = screenToHex(e.clientX - r.left, e.clientY - r.top);
    const changed = (h && (!hovered || h.x !== hovered.x || h.y !== hovered.y)) || (!h && hovered);
    hovered = h;
    if (h) updateHUD(tiles[h.y][h.x]);
    if (changed) render();
  }
});
canvas.addEventListener("click", e => {
  if (moved > 6) return;
  const r = canvas.getBoundingClientRect();
  const h = screenToHex(e.clientX - r.left, e.clientY - r.top);
  selected = h;
  if (h) updateHUD(tiles[h.y][h.x]);
  render();
});
canvas.addEventListener("wheel", e => {
  e.preventDefault();
  const r = canvas.getBoundingClientRect();
  const mx = e.clientX - r.left, my = e.clientY - r.top;
  const worldX = (mx + camX) / hexSize, worldY = (my + camY) / hexSize;
  hexSize = Math.max(16, Math.min(110, hexSize * (e.deltaY < 0 ? 1.15 : 1 / 1.15)));
  camX = worldX * hexSize - mx;
  camY = worldY * hexSize - my;
  render();
}, { passive: false });

let touchDist = 0;
canvas.addEventListener("touchstart", e => {
  if (e.touches.length === 1) { lastX = e.touches[0].clientX; lastY = e.touches[0].clientY; }
  else if (e.touches.length === 2) {
    touchDist = Math.hypot(e.touches[0].clientX - e.touches[1].clientX,
                           e.touches[0].clientY - e.touches[1].clientY);
  }
}, { passive: true });
canvas.addEventListener("touchmove", e => {
  e.preventDefault();
  if (e.touches.length === 1) {
    camX -= e.touches[0].clientX - lastX;
    camY -= e.touches[0].clientY - lastY;
    lastX = e.touches[0].clientX; lastY = e.touches[0].clientY;
  } else if (e.touches.length === 2) {
    const d = Math.hypot(e.touches[0].clientX - e.touches[1].clientX,
                         e.touches[0].clientY - e.touches[1].clientY);
    hexSize = Math.max(16, Math.min(110, hexSize * (d / (touchDist || d))));
    touchDist = d;
  }
  render();
}, { passive: false });
