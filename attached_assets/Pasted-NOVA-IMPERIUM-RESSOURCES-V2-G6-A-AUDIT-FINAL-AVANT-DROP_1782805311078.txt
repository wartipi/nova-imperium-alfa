ÉTAPE 2 — AUDIT DB GLOBAL

Exécuter SELECT de contrôle :

1. Ressources V1 neutralisées

SELECT 'player_bank' AS table_name,
COALESCE(SUM(gold),0) AS gold,
COALESCE(SUM(iron),0) AS iron,
COALESCE(SUM(copper),0) AS copper,
COALESCE(SUM(fur),0) AS fur
FROM player_bank;

Répéter pour :

* player_transport
* player_market_box
* city_inventory
* city_pending_harvest

2. market_fee_box

SELECT
COALESCE(SUM(gold),0) AS gold,
COALESCE(SUM(fracten),0) AS fracten
FROM market_fee_box;

3. cities per-turn

SELECT
COALESCE(SUM(gold_per_turn),0) AS gold_per_turn,
COALESCE(SUM(iron_per_turn),0) AS iron_per_turn,
COALESCE(SUM(copper_per_turn),0) AS copper_per_turn,
COALESCE(SUM(fur_per_turn),0) AS fur_per_turn,
COALESCE(SUM(fracten_per_turn),0) AS fracten_per_turn,
COALESCE(SUM(common_metals_per_turn),0) AS common_metals_per_turn,
COALESCE(SUM(leather_fur_per_turn),0) AS leather_fur_per_turn
FROM cities;

4. market_trades

SELECT
COUNT(*) AS trade_count,
COALESCE(SUM(total_gold),0) AS total_gold_sum,
COALESCE(SUM(total_fracten),0) AS total_fracten_sum,
SUM(CASE WHEN total_gold = total_fracten THEN 0 ELSE 1 END) AS mismatches
FROM market_trades;

5. market_orders legacy

SELECT resource_type, status, COUNT(*) AS count
FROM market_orders
WHERE resource_type IN ('gold','iron','copper','fur')
GROUP BY resource_type, status
ORDER BY resource_type, status;
